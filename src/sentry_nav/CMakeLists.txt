cmake_minimum_required(VERSION 3.16)
project(sentry_nav LANGUAGES C CXX)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp rclcpp_action rclcpp_lifecycle nav2_core nav2_costmap_2d nav2_msgs
  nav2_behavior_tree behaviortree_cpp_v3 nav2_util std_srvs lifecycle_msgs
  pluginlib tf2_ros tf2_geometry_msgs nav_msgs geometry_msgs std_msgs visualization_msgs action_msgs sentry_nav_interfaces sentry_omni_kino_planner)
  find_package(${dep} REQUIRED)
endforeach()
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(ENABLE_MKL_PARDISO OFF CACHE BOOL "" FORCE)
set(PRINTING OFF CACHE BOOL "" FORCE)
set(CTRLC OFF CACHE BOOL "" FORCE)
set(DFLOAT OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/osqp EXCLUDE_FROM_ALL)
add_library(tdt_minimum_snap STATIC vendor/tdt_minimum_snap/minimumSnap.cpp vendor/tdt_minimum_snap/sfcSquare.cpp)
target_compile_features(tdt_minimum_snap PUBLIC cxx_std_17)
target_include_directories(tdt_minimum_snap PUBLIC vendor/tdt_minimum_snap)
target_link_libraries(tdt_minimum_snap PUBLIC Eigen3::Eigen PRIVATE osqpstatic)
add_library(tinympc STATIC vendor/tinympc/admm.cpp vendor/tinympc/tiny_api.cpp vendor/tinympc/rho_benchmark.cpp)
target_compile_features(tinympc PUBLIC cxx_std_17)
target_include_directories(tinympc PUBLIC vendor)
target_link_libraries(tinympc PUBLIC Eigen3::Eigen)
add_library(sentry_mpc STATIC src/mpc.cpp src/trajectory.cpp src/minimum_snap.cpp)
target_compile_features(sentry_mpc PUBLIC cxx_std_20)
target_include_directories(sentry_mpc PUBLIC include)
target_link_libraries(sentry_mpc PUBLIC sentry_omni_kino_planner::omni_kino_core Eigen3::Eigen PRIVATE tinympc tdt_minimum_snap)
add_library(sentry_nav_plugins SHARED src/plugins.cpp src/translation_goal_checker.cpp)
set_target_properties(sentry_nav_plugins PROPERTIES SKIP_BUILD_RPATH TRUE)
target_compile_features(sentry_nav_plugins PUBLIC cxx_std_20)
target_include_directories(sentry_nav_plugins PUBLIC include)
target_link_libraries(sentry_nav_plugins sentry_mpc Threads::Threads)
ament_target_dependencies(sentry_nav_plugins rclcpp rclcpp_lifecycle nav2_core nav2_costmap_2d
  pluginlib tf2_ros tf2_geometry_msgs nav_msgs geometry_msgs std_msgs visualization_msgs sentry_nav_interfaces)
add_library(sentry_ordered_navigation_bt SHARED src/ordered_navigation_bt.cpp)
target_compile_features(sentry_ordered_navigation_bt PUBLIC cxx_std_20)
target_compile_definitions(sentry_ordered_navigation_bt PRIVATE BT_PLUGIN_EXPORT)
ament_target_dependencies(sentry_ordered_navigation_bt nav2_behavior_tree behaviortree_cpp_v3 sentry_nav_interfaces)
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
add_library(sentry_passable_layer SHARED src/passable_layer.cpp)
target_compile_features(sentry_passable_layer PUBLIC cxx_std_17)
target_link_libraries(sentry_passable_layer yaml-cpp)
ament_target_dependencies(sentry_passable_layer nav2_costmap_2d pluginlib tf2_ros tf2_geometry_msgs)
pluginlib_export_plugin_description_file(nav2_costmap_2d costmap_plugins.xml)
install(TARGETS sentry_passable_layer LIBRARY DESTINATION lib)
install(PROGRAMS scripts/passable_editor.py DESTINATION lib/${PROJECT_NAME})
add_executable(sentry_motion_gate src/motion_gate.cpp)
target_compile_features(sentry_motion_gate PRIVATE cxx_std_20)
target_link_libraries(sentry_motion_gate yaml-cpp)
ament_target_dependencies(sentry_motion_gate rclcpp rclcpp_action nav2_msgs sentry_nav_interfaces
  tf2_ros tf2_geometry_msgs nav_msgs geometry_msgs)
ament_target_dependencies(sentry_motion_gate action_msgs)
add_executable(sentry_navigation src/navigation_runtime.cpp)
target_compile_features(sentry_navigation PRIVATE cxx_std_20)
target_link_libraries(sentry_navigation sentry_nav_plugins Threads::Threads)
ament_target_dependencies(sentry_navigation rclcpp rclcpp_action rclcpp_lifecycle nav2_msgs
  nav2_core nav2_costmap_2d nav2_util tf2_ros tf2_geometry_msgs std_msgs std_srvs lifecycle_msgs
  sentry_nav_interfaces)
add_executable(sentry_route_manager src/route_manager.cpp)
target_compile_features(sentry_route_manager PRIVATE cxx_std_20)
target_link_libraries(sentry_route_manager Threads::Threads)
ament_target_dependencies(sentry_route_manager rclcpp rclcpp_action rcl_interfaces nav2_msgs
  sentry_nav_interfaces std_msgs std_srvs tf2_ros)
foreach(target sentry_navigation sentry_mpc sentry_nav_plugins sentry_motion_gate sentry_route_manager)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
install(TARGETS sentry_nav_plugins sentry_ordered_navigation_bt LIBRARY DESTINATION lib)
install(TARGETS sentry_motion_gate sentry_navigation sentry_route_manager DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY vendor/ DESTINATION share/${PROJECT_NAME}/licenses
  FILES_MATCHING PATTERN "LICENSE*" PATTERN "NOTICE*")
ament_package()
