cmake_minimum_required(VERSION 3.16)
project(sentry_nav_rviz)
find_package(ament_cmake REQUIRED)
foreach(dep rviz_common pluginlib rclcpp rclcpp_action nav2_msgs geometry_msgs visualization_msgs std_msgs std_srvs sentry_nav_interfaces)
  find_package(${dep} REQUIRED)
endforeach()
find_package(Qt5 REQUIRED COMPONENTS Widgets)
set(CMAKE_AUTOMOC ON)
add_library(sentry_navigation_panel SHARED src/navigation_panel.cpp include/sentry_nav_rviz/navigation_panel.hpp)
target_compile_features(sentry_navigation_panel PUBLIC cxx_std_17)
target_include_directories(sentry_navigation_panel PUBLIC include)
target_link_libraries(sentry_navigation_panel Qt5::Widgets)
ament_target_dependencies(sentry_navigation_panel rviz_common pluginlib rclcpp rclcpp_action
  nav2_msgs geometry_msgs visualization_msgs std_msgs std_srvs sentry_nav_interfaces)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS sentry_navigation_panel LIBRARY DESTINATION lib)
ament_package()
